cmake_minimum_required(VERSION 3.20)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(InlayEngine LANGUAGES C CXX)
endif()

if(NOT APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Inlay's server supports macOS and Linux.")
endif()

# Keep the speech runtime's ggml independent from the text helper's version.
option(INLAY_CUDA "Use CUDA for Linux inference" OFF)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
if(APPLE)
    set(GGML_METAL ON CACHE BOOL "" FORCE)
    set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
    set(GGML_ACCELERATE ON CACHE BOOL "" FORCE)
    set(GGML_BLAS ON CACHE BOOL "" FORCE)
    set(GGML_BLAS_VENDOR Apple CACHE STRING "" FORCE)
    set(GGML_CUDA OFF CACHE BOOL "" FORCE)
else()
    set(GGML_METAL OFF CACHE BOOL "" FORCE)
    set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
    set(GGML_BLAS OFF CACHE BOOL "" FORCE)
    set(GGML_CUDA ${INLAY_CUDA} CACHE BOOL "" FORCE)
    # Static cudart/cublas so the tarball does not need /usr/local/cuda.
    # libcuda.so.1 still comes from the host driver. Do not pull in NCCL.
    if(INLAY_CUDA)
        set(GGML_STATIC ON CACHE BOOL "" FORCE)
        set(GGML_CUDA_NCCL OFF CACHE BOOL "" FORCE)
    endif()
endif()
find_package(Threads REQUIRED)

set(WHISPER_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/whisper.cpp")
if(NOT EXISTS "${WHISPER_SOURCE}/include/whisper.h")
    message(FATAL_ERROR "Run git submodule update --init --recursive first.")
endif()

# Inlay's changes to the speech runtime live in Engine/patches. They apply to a
# copy in the build tree, so the submodule stays an unmodified upstream checkout.
file(GLOB INLAY_WHISPER_PATCHES "${CMAKE_CURRENT_SOURCE_DIR}/patches/*.patch")
list(SORT INLAY_WHISPER_PATCHES)
if(INLAY_WHISPER_PATCHES)
    find_program(INLAY_PATCH_TOOL patch REQUIRED)
    # Recopy when a patch or any file it touches changes (e.g. a submodule bump).
    set(stamp "")
    foreach(patch_file IN LISTS INLAY_WHISPER_PATCHES)
        file(SHA256 "${patch_file}" patch_hash)
        string(APPEND stamp "${patch_hash}\n")
        file(STRINGS "${patch_file}" touched REGEX "^\\+\\+\\+ b/")
        foreach(line IN LISTS touched)
            string(REGEX REPLACE "^\\+\\+\\+ b/" "" relative "${line}")
            if(EXISTS "${WHISPER_SOURCE}/${relative}")
                file(SHA256 "${WHISPER_SOURCE}/${relative}" source_hash)
                string(APPEND stamp "${relative} ${source_hash}\n")
            endif()
        endforeach()
    endforeach()
    set(patched "${CMAKE_CURRENT_BINARY_DIR}/whisper-patched")
    set(previous "")
    if(EXISTS "${patched}/.inlay-patches")
        file(READ "${patched}/.inlay-patches" previous)
    endif()
    if(NOT previous STREQUAL stamp)
        file(REMOVE_RECURSE "${patched}")
        file(COPY "${WHISPER_SOURCE}/" DESTINATION "${patched}" PATTERN ".git" EXCLUDE)
        foreach(patch_file IN LISTS INLAY_WHISPER_PATCHES)
            execute_process(
                COMMAND "${INLAY_PATCH_TOOL}" -p1 --batch --forward --quiet -i "${patch_file}"
                WORKING_DIRECTORY "${patched}"
                RESULT_VARIABLE patch_result)
            if(NOT patch_result EQUAL 0)
                message(FATAL_ERROR "Could not apply ${patch_file} to vendor/whisper.cpp. Update the patch for this submodule version.")
            endif()
        endforeach()
        file(WRITE "${patched}/.inlay-patches" "${stamp}")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${INLAY_WHISPER_PATCHES})
    set(WHISPER_SOURCE "${patched}")
endif()
add_subdirectory("${WHISPER_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/whisper")

add_executable(inlay-engine worker.cpp)
target_compile_features(inlay-engine PRIVATE cxx_std_17)
target_compile_options(inlay-engine PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(inlay-engine SYSTEM PRIVATE "${WHISPER_SOURCE}/examples")
target_link_libraries(inlay-engine PRIVATE parakeet whisper Threads::Threads)
set_target_properties(inlay-engine PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/Engine")
